class Solution {
    public int lengthOfLongestSubstring(String s) {
        ArrayList<Character> elements=new ArrayList<Character>();
        int start=0;
        int end=0;
        int maxCount=0;

        if(s.length() <=1){
            return s.length();
        }
        while(end<s.length() && start<s.length()){
            if(elements.contains(s.charAt(end))){
                maxCount=Math.max(maxCount,elements.size());
                elements.remove(0);
                start++;
            }else{
                elements.add(s.charAt(end));
                end++;
            }
        }

        return maxCount=Math.max(maxCount,elements.size());
        
        
    }
}

class Solution {
    public boolean isValid(String s) {
        Stack<Character> brackets=new Stack<Character>();
        ArrayList<Character> openBracks=new ArrayList<Character>();
        openBracks.add('{');
        openBracks.add('(');
        openBracks.add('[');
        openBracks.add('}');
        openBracks.add(')');
        openBracks.add(']');

        for(int i=0;i<s.length();i++){
            if(openBracks.indexOf(s.charAt(i)) < 3){
                brackets.push(s.charAt(i));
            }else{
                int val= openBracks.indexOf(s.charAt(i));
                if(brackets.isEmpty()){
                    return false;
                }
                int top=openBracks.indexOf(brackets.pop());
                if(top+3==val){
                    continue;
                }else{
                    return false;
                }
            }

        }


        return brackets.isEmpty() ?  true: false;
        
    }
}

Q4.Longest Repeating character replacement
class Solution {
    public int characterReplacement(String s, int k) {
        int len = s.length();
        int left = 0;
        int right = 0;
        AtomicInteger maxFreq=new AtomicInteger(0);
        int ans = 0;
        Map<Character, Integer> count = new HashMap<>();

        while (right < len) {
            char current = s.charAt(right);
            int substrlen = right - left + 1;

            
            count.put(current, count.getOrDefault(current, 0) + 1);

            count.forEach((key,value)->{
                maxFreq.set(Math.max(maxFreq.get(),value));

            });

            
           

            
            if (substrlen - maxFreq.get() <= k) {
                ans = Math.max(ans, substrlen);
                right++;
                continue;
            }

            
            char leftChar = s.charAt(left);
            if (count.get(leftChar) == 1) {
                count.remove(leftChar);
            } else {
                count.replace(leftChar, count.get(leftChar) - 1);
            }
            left++;
            right++;
        }

        return ans;

    }
}

q8. minimum window substring
class Solution {
    public String minWindow(String s, String t) {

        int lengthS = s.length();
        int lengthT = t.length();

        // checks for the boundary values cases 
        if (lengthT == 0 || lengthS < lengthT) {
            return "";
        }

        // stores the count of character in string t
        Map<Character, Integer> needMap = new HashMap<Character, Integer>();
     
        for (int i = 0; i < lengthT; i++) {
            needMap.put(t.charAt(i), needMap.getOrDefault(t.charAt(i), 0) + 1);
        }

        // stores the count of characters in window 

        Map<Character, Integer> window = new HashMap<Character, Integer>();
        // how many conditons we have statsified 
        int have = 0;
        // we are focusing on how many unique characters are there to make reuqired string 
        int need = needMap.size();
        int strLen = Integer.MAX_VALUE;
        String str = "";
        int left = 0;
        int right = 0;

        for (; right < lengthS; right++) {
            // current element check
            Character current = s.charAt(right);
            window.put(current, window.getOrDefault(current, 0) + 1);
            
            // if the current element is part of string t and if we have enough current which are required to make t
            if (needMap.containsKey(current) && window.get(current).equals(needMap.get(current))) {
                have += 1;
            }
            
            // to find min string we remove use less character not part of string t and if left char is part of string then we need to check have constion
            while (have == need) {
                // min string 
                int sbstrLen = right - left + 1;
                if (sbstrLen < strLen) {
                    strLen = sbstrLen;
                    str = s.substring(left, right + 1);
                }
                
                // left character
                Character c = s.charAt(left);
                if (window.get(c) <= 1) {
                    // left char get zero avoid -1 
                    window.put(c, 0);
                } else {
                    // left char value decrement 
                    window.put(c, window.get(c) - 1);
                }
                // check weather the c is enough to make reuired string or not 
                if (needMap.containsKey(c) && window.get(c) < needMap.get(c)) {
                    have--;
                }
                // left char removed
                left++;
            }
        }

        return str;

    }
}

        ArrayList<String> ansList=new ArrayList<String>();

            //  i points to the start of number and j points to #

                j++;
            }
            
            // length of word

q12.merge  k sorted linked lists
/**
 * Definition for singly-linked list.
 * public class ListNode {
 * int val;
 * ListNode next;
 * ListNode() {}
 * ListNode(int val) { this.val = val; }
 * ListNode(int val, ListNode next) { this.val = val; this.next = next; }
 * }
 */
class Solution {
    public ListNode mergeKLists(ListNode[] lists) {
        PriorityQueue<Integer> nodeValue = new PriorityQueue<Integer>();

        // add the all elements in priority queue
        for (int i = 0; i < lists.length; i++) {
            ListNode current = lists[i];
            while (current != null) {
                nodeValue.add(current.val);
                current = current.next;
            }
        }

        // head of ans
        ListNode head = new ListNode();
        if (nodeValue.isEmpty()) {
            return null;
        }
        head.val = nodeValue.poll();
        ListNode last = head;

        // priority queue auto sort the content and pop the min element of currnet array
        while (!nodeValue.isEmpty()) {
            ListNode temp = new ListNode();
            temp.val = nodeValue.poll();
            last.next = temp;
            last = temp;
        }

        return head;

    }
}




q13. Top K Frequent Elements
class Solution {
    public int[] topKFrequent(int[] nums, int k) {
        if (nums.length == 1) {
            return nums;
        }
        // sort the elements first  to enter one element only once 
        Arrays.sort(nums);
        int[] ans = new int[k];
        // keep the note of all  unique counts
        PriorityQueue<Integer> countCompare = new PriorityQueue<Integer>(Collections.reverseOrder());
        // key : count --> value : elements that possess that count
        Map<Integer, ArrayList<Integer>> countValue = new HashMap<Integer, ArrayList<Integer>>();
        // first element 
        int last = nums[0];
        int count = 0;
        for (int i = 0; i < nums.length; i++) {
            if (nums[i] == last) {
                count++;
            } else {
                // already exhist count try to group all the elements who possess same counts 
                if (countValue.containsKey(count)) {
                    ArrayList<Integer> temp = countValue.get(count);
                    temp.add(last);
                    countValue.put(count, temp);
                } else {
                    // new count new group
                    ArrayList<Integer> temp = new ArrayList<Integer>();
                    temp.add(last);
                    countValue.put(count, temp);
                }
                // new count 
                if (!countCompare.contains(count)) {
                    countCompare.add(count);
                }
                count = 1;
                last = nums[i];
            }

        }

       
        // last element case 
        if (count != 0) {
            if (countValue.containsKey(count)) {
                ArrayList<Integer> temp = countValue.get(count);
                temp.add(last);
                countValue.put(count, temp);
            } else {
                ArrayList<Integer> temp = new ArrayList<Integer>();
                temp.add(last);
                countValue.put(count, temp);
            }
            if (!countCompare.contains(count)) {
                countCompare.add(count);
            }

        }
        

        // reading out the data 
        int index = 0;
        while (index < k && !countCompare.isEmpty()) {
            int elem = countCompare.poll();
            // all elements of same count 
            ArrayList<Integer> elements = countValue.get(elem);
            int curr = 0;
            // we need only unique elements whos possess same count with max elem k
            while (index < k && curr < elements.size()) {
                ans[index] = elements.get(curr);
                index++;
                curr++;
            }

        }

        return ans;

    }
}

class MedianFinder {
    // in order to solve this problem we make two maxHeap minheap
    //  which follows two conditon 
    // cond-1 : size difference between both heap should be 0-1
    // because if totalNum is odd - then heap with larger size return the mid elem
    //  if totalNum  is even - then both heap will get automaticall eqauly item and both peek return mid elem respective of their heap
    //  condtion-2 : all elem of minHeap should be greater than maxHeap
    
    // small heap is max-heap
    PriorityQueue<Integer> maxHeap=new PriorityQueue<Integer>(Collections.reverseOrder());
    //  large heap is min-heap
    PriorityQueue<Integer> minHeap=new PriorityQueue<Integer>();

    public MedianFinder() {
        
    }
    

    //  condition 1  check  size length 
    public void lengthCheck(){
         while( Math.abs(minHeap.size()-maxHeap.size()) > 1){
            if(minHeap.size() > maxHeap.size()){
                int a = minHeap.poll();
                maxHeap.add(a);
            }else{
                int a =maxHeap.poll();
                minHeap.add(a);
            }
        }
    }
    
    public void addNum(int num) {
        maxHeap.add(num);
        // condition 2 checked wheather all elem of minHeap is greater from all elements of maxHeap
        lengthCheck();
        while(!minHeap.isEmpty() && !maxHeap.isEmpty() && minHeap.peek() < maxHeap.peek()){
            int a = maxHeap.poll();
            minHeap.add(a);
        }
        lengthCheck();  
    }
    
    public double findMedian() {
        // numebr of elem even
        if(minHeap.size()==maxHeap.size()){
            double val=((minHeap.peek() + maxHeap.peek())/2.0);  
            return val;
        }
        //  number of elem odd
        return minHeap.size()>maxHeap.size() ? minHeap.peek() : maxHeap.peek() ;
        
    }
}

class Solution {
    public int maxProduct(int[] nums) {
        int prefix=1;
        int suffix=1;
        int ans=Integer.MIN_VALUE;
        for(int i=0;i<nums.length;i++){
            if(prefix==0) prefix =1;
            if(suffix==0) suffix =1;
            prefix=prefix*nums[i];
            suffix=suffix*nums[nums.length-i-1];
            ans=Math.max(ans,Math.max(prefix,suffix));
        }
        return ans;
        
    }
}

class Solution {
    public int[] productExceptSelf(int[] nums) {
        int len=nums.length;
        
        int [] prefixProduct=new int[len];
        int [] suffixProduct=new int[len];
        int prefix=1;
        int suffix=1;
        for(int i=0;i<len;i++){
            prefix=prefix*nums[i];
            suffix=suffix*nums[len-i-1];
            prefixProduct[i]=prefix;
            suffixProduct[len-i-1]=suffix;
        }

        int [] ans=new int[len];
        for(int i=0;i<len;i++){
            int left= i==0 ? 1 : prefixProduct[i-1];
            int right= i==len-1 ? 1 : suffixProduct[i+1] ;
            ans[i]=left*right;
        }

        return ans;


        
    }
}

q20. Climb stairs
class Solution {
    // memoization
    int [] dp;

    public int downStairs(int index){
        // reached at last index
        if(index==0) return 1;
        // reacehd at second last index index so only one jump allowed
        if(index==1) return 1;
        
        // arleady solved
        if(dp[index]!=-1) return dp[index];
        // one step down
        int oneStep=downStairs(index-1);
        // two step down
        int twoStep=downStairs(index-2);
        // total way
        dp[index]=oneStep+twoStep;
        return dp[index];
    }
    public int climbStairs(int n) {
        // there are 5 stairs then 5->4->3->2->1->0
        dp=new int[n+1];
        Arrays.fill(dp,-1);

        return downStairs(n);
        
    }
}

class Solution {
    int [][] dp;

    public int robbery(int [] nums,int currHouse,int last){
        // reached at the last house
        if(currHouse==0){
            if(last==0){
              // second last home not picked
              return nums[0];
            }
            // second last home picked
            return 0;
        }

        if(dp[currHouse][last]!=-1){
            return dp[currHouse][last];
        }
        int pick=0;
        if(last==0){
            //  last house not picked
            pick=nums[currHouse]+robbery(nums,currHouse-1,1);
        }
        // last house picked
        int notPick=robbery(nums,currHouse-1,0);

        // max robbery
        dp[currHouse][last]=Math.max(pick,notPick);

        return dp[currHouse][last];

    }
    public int rob(int[] nums) {
        int len=nums.length;
        dp=new int [len][2];
        for(int i=0;i<len;i++){
            Arrays.fill(dp[i],-1);
        }

        return robbery(nums,len-1,0);
        
    }
}

class Solution {

    public int [][] dp;

    public int robotMove(int m,int n,int currRow,int currCol){
        if(currRow <0 || currCol<0){
            return 0;
        }
        if(currRow==0 && currCol==0){
            return 1;
        }

        if(dp[currRow][currCol]!=-1){
            return dp[currRow][currCol];
        }

        int topMov=robotMove(m,n,currRow-1,currCol);
        int leftMov=robotMove(m,n,currRow,currCol-1);


        return dp[currRow][currCol]=topMov+leftMov;

    }
    public int uniquePaths(int m, int n) {
        dp=new int[m][n];
        for(int i=0;i<m;i++){
            Arrays.fill(dp[i],-1);
        }

        return robotMove(m,n,m-1,n-1);
        
    }
}

class Solution {

    int [][] dp;

    public int commonSeq(String text1,String text2,int str1,int str2){
        if(str1<0 || str2<0){
            return 0;
        }

        if(dp[str1][str2]!=-1){
            return dp[str1][str2];
        }

        int same=0;
        if(text1.charAt(str1) == text2.charAt(str2)){
            same=1+commonSeq(text1,text2,str1-1,str2-1);
        }

        int left=commonSeq(text1,text2,str1-1,str2);
        int right=commonSeq(text1,text2,str1,str2-1);

        dp[str1][str2]=Math.max(same,Math.max(left,right));
        return dp[str1][str2];        
    }
    public int longestCommonSubsequence(String text1, String text2) {

            Arrays.fill(dp2[i],-1);

class Solution {

    int [][] dp;
    public int increasingSubs(int [] nums,int last,int current){
        if(nums.length==1){
            return 1;
        }
        if(current==nums.length){
            return 0;
        }
        
        // increament to get the correct value
        if( dp[current][last+1]!=-1){
            return dp[current][last+1];
        }


        int notPick=increasingSubs(nums,last,current+1);
        int pick=0;
        // in case of no elem is picked untill now 
        if(last==-1 || nums[current]>nums[last]){
            pick=1+increasingSubs(nums,current,current+1);
        }

        dp[current][last+1]=Math.max(notPick,pick);
        return dp[current][last+1];
    }
    public int lengthOfLIS(int[] nums) {
        // initilally no elemnet is selected which marked as last as -1 
        // to have access we stores it in last +1
        // dp[current][last] actually stores at dp[current][last+1]
        dp=new int[nums.length][nums.length+1];
        for(int i=0;i<nums.length;i++){
            Arrays.fill(dp[i],-1);

        }
        
        // so intiallly no last item selected
        return increasingSubs(nums,-1,0);
        
    }
}




q26.Word break problem
class Solution {

    int [] dp;

    public int breaks(String s,int current,int length,List<String> wordDict){
        //  we are only moving forward when we have true in back string
        if(current>=length){
            return 1;
            
        }
        
        // memoization
        if(dp[current]!=-1){
            return dp[current];
        }


        // check wheather we have full word or not 
        if(wordDict.contains(s.substring(current,length))){
            return dp[current]=1;

        }
        
        // checking for the variable length string
        for(int i=current+1;i<=length;i++){
            String temp=s.substring(current,i);
            // checking wheather dictionary have futre words as well or not 
            if(wordDict.contains(temp) && breaks(s,i,length,wordDict) ==1){
                return dp[current]=1;
            }
        }

        return dp[current]=0;
    }
    public boolean wordBreak(String s, List<String> wordDict) {
        dp =new int[s.length()+1];
        Arrays.fill(dp,-1);
        return breaks(s,0,s.length(),wordDict) == 1 ? true : false;
        
    }
}

        return dp[curr_target] = curr;
    }

    }

class Solution {

    int[] dp;

    public int jumps(int[] nums, int current) {
        if (current >= nums.length - 1) {
            return 1;
        }

        if (dp[current] != -1) {
            return dp[current];
        }

        for (int i = 1; i <= nums[current]; i++) {
            if (jumps(nums, current + i) == 1) {
                return dp[current] = 1;

            }
        }

        return dp[current] = 0;

    }

    public boolean canJump(int[] nums) {
        dp = new int[nums.length];
        Arrays.fill(dp, -1);
        return jumps(nums, 0) == 1 ? true : false;

    }
}

    }

        return ans >= (int) 1e7 ? -1 : ans;
        
    }

q36.Word search
class Solution {
    int[][] traversal;

    public boolean matrixMove(char[][] board, String word, int current, int x, int y) {
        // whole word traversal
        if (current >= word.length()) {
            return true;
        }

        // out of bound condition
        if (x < 0 || y < 0 || x >= board.length || y >=board[0].length) {
            return false;
        }

        // word does not match
        // -1 means traversal already
        if (board[x][y] != word.charAt(current) || traversal[x][y] == -1) {
            return false;
        }

        // forward tracing
        traversal[x][y] = -1;
        // movements
        boolean left = matrixMove(board, word, current + 1, x, y - 1);
        boolean right = matrixMove(board, word, current + 1, x, y + 1);
        boolean top = matrixMove(board, word, current + 1, x - 1, y);
        boolean bottom = matrixMove(board, word, current + 1, x + 1, y);
        // backward tracing
        traversal[x][y] = 0;

        return left || right || top || bottom;

    }

    public boolean exist(char[][] board, String word) {
        int row=board.length;
        int col=board[0].length;
        traversal = new int[row][col];
        for(int i=0;i<row;i++){
            Arrays.fill(traversal[i],0);
        }

        

        for(int i=0;i<row;i++){
            
            for(int j=0;j<col;j++){
                boolean val=matrixMove(board,word,0,i,j);
                if(val) return true;
            }
        }

        return false;

    }
}

class Solution {
    public int eraseOverlapIntervals(int[][] intervals) {
        ArrayList<ArrayList<Integer>> intvList=new ArrayList<ArrayList<Integer>>();
        for(int i=0;i<intervals.length;i++){
            ArrayList temp=new ArrayList<Integer>();
            for(int j=0;j<intervals[i].length;j++){
                temp.add(intervals[i][j]);
            }
            intvList.add(temp);
        }


        Comparator<ArrayList<Integer>> cmp=new Comparator<ArrayList<Integer>>(){
            public int compare(ArrayList<Integer> a,ArrayList<Integer> b){
                if(a.get(0)==b.get(0)){
                    return Integer.compare(a.get(1),b.get(1));
                }

                return Integer.compare(a.get(0),b.get(0));
            }
        };

        Collections.sort(intvList,cmp);


        int endPoint = intvList.get(0).get(1);
        int changes=0;

        for(int i = 1;i<intvList.size();i++){
            int x=intvList.get(i).get(0);
            int y=intvList.get(i).get(1);

            if(x>=endPoint){
                // no overlapping
                endPoint = y;
            }else{
                // overlapping
                changes++;
                endPoint=Math.min(y,endPoint);

            }
        }
       



        return changes;
        
    }
}

                }

            }

q39.Meeting Rooms II
/**
 * Definition of Interval:
 * public class Interval {
 *     public int start, end;
 *     public Interval(int start, int end) {
 *         this.start = start;
 *         this.end = end;
 *     }
 * }
 */

class Solution {
    public int minMeetingRooms(List<Interval> intervals) {
        int size=intervals.size();
        // from pictorial representation noticing the timeline
        int [] start=new int[size];
        int [] end=new int[size];
        for(int i=0;i<size;i++){
            start[i]=intervals.get(i).start;
            end[i]=intervals.get(i).end;
        }
        Arrays.sort(start);
        Arrays.sort(end);

        int s=0,e=0;
        int res=0,count=0;
        
        // alloting rooms
        while(s<size){
            if(start[s] < end[e]) {
                // new meeting started before an meeting ends
                // requrie more rooms
                count++;
                s++;
            }else{
                // meeting ends but not handling the meeting whose start stime is start[s] 
                // start[s] handled in next meeting
                // less rooms requirement
                count--;
                e++;
            }
            res=Math.max(res,count);
        }


        return res;
    }
}
