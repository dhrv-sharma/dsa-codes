q1. find k largest element
class Solution {
    public int findKthLargest(int[] nums, int k) {

        PriorityQueue<Integer> numsSort=new PriorityQueue<Integer>(Collections.reverseOrder());
        for(int i=0;i<nums.length;i++){
            numsSort.add(nums[i]);
        }
        for(int i=0;i<k-1;i++){
            numsSort.poll();
        }

        return numsSort.poll();
        
    }
}
